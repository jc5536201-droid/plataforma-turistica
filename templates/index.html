function actualizarPanel(data) { const panel = document.getElementById('panelRuta'); const nombresCriterio = { distancia: 'Distancia mínima', tiempo: 'Tiempo mínimo', costo: 'Costo (métrica informativa — equivale a distancia)' }; const nombreCriterio = nombresCriterio[data.criterio] || data.criterio || 'Criterio seleccionado'; const badgeMetricas = ` Métricas: OSRM `; const tCond = Number(data.tiempo_conduccion_min || data.tiempo_min || 0); const tTotal = Number(data.tiempo_min || tCond); const holgura = tTotal - tCond; let desgloseTiempo = ""; if (holgura > 0.1) { desgloseTiempo = `
${tCond.toFixed(0)} min conducción + ${holgura.toFixed(0)} min holgura (×${CONFIG.factorHolgura})
`; } let html = `
Ruta óptima Dijkstra
${badgeMetricas}
Criterio: ${nombreCriterio}
${Number(data.distancia_km).toFixed(2)} km ${Number(data.tiempo_min).toFixed(0)} min $${Number(data.costo).toFixed(2)}
${desgloseTiempo}

Origen: ${data.nodos_ruta?.[0]?.nombre || 'No disponible'}

Destino: ${data.nodos_ruta?.[data.nodos_ruta.length - 1]?.nombre || 'No disponible'}

Camino óptimo (Dijkstra)
`; if (Array.isArray(data.nodos_ruta)) { data.nodos_ruta.forEach((nodo, indice) => { html += `
${indice + 1}
${nodo.nombre}
Nodo: ${nodo.cod}
`; }); } if (Array.isArray(data.segmentos) && data.segmentos.length > 1) { html += `
Detalle de los tramos
`; data.segmentos.forEach((tramo, indice) => { const origen = data.nodos_ruta?.find(n => Number(n.id) === Number(tramo.origen)); const destino = data.nodos_ruta?.find(n => Number(n.id) === Number(tramo.destino)); html += `
${indice + 1}. ${origen?.nombre || 'Origen'} → ${destino?.nombre || 'Destino'}
${Number(tramo.distancia_km).toFixed(2)} km · ${Number(tramo.tiempo_min).toFixed(1)} min · $${Number(tramo.costo).toFixed(2)}
`; }); } panel.innerHTML = html; }
